// app.js
let masterPartsList = [];
let logs = [];
let lastScannedBarcode = '';

// Load master_parts_list.json
fetch('master_parts_list.json')
	.then(res => res.json())
	.then(data => { masterPartsList = data; })
	.catch(() => { masterPartsList = []; });

function findPartByBarcode(barcode) {
	return masterPartsList.find(part => part.Barcode === barcode);
}

function updatePartDetails(part) {
	document.getElementById('part-name').textContent = part ? part.PartName : '';
	document.getElementById('part-desc').textContent = part ? part.Description : '';
	document.getElementById('part-stock').textContent = part ? part.StockLevel : '';
}

function updateLogsTable() {
	const tbody = document.getElementById('logs-table-body');
	tbody.innerHTML = '';
	logs.forEach(log => {
		const tr = document.createElement('tr');
		tr.innerHTML = `
			<td class="border px-2 py-1">${log.timestamp}</td>
			<td class="border px-2 py-1">${log.woNumber}</td>
			<td class="border px-2 py-1">${log.barcode}</td>
			<td class="border px-2 py-1">${log.partName}</td>
			<td class="border px-2 py-1">${log.description}</td>
			<td class="border px-2 py-1">${log.stockLevel}</td>
		`;
		tbody.appendChild(tr);
	});
}

function exportLogsToCSV() {
	if (!logs.length) return;
	const headers = ['Timestamp','WO Number','Barcode','Part Name','Description','Stock Level'];
	const rows = logs.map(log => [log.timestamp, log.woNumber, log.barcode, log.partName, log.description, log.stockLevel]);
	let csvContent = headers.join(',') + '\n' + rows.map(r => r.map(x => '"'+(x||'')+'"').join(',')).join('\n');
	const blob = new Blob([csvContent], { type: 'text/csv' });
	const url = URL.createObjectURL(blob);
	const a = document.createElement('a');
	a.href = url;
	a.download = 'assigned_parts_log.csv';
	document.body.appendChild(a);
	a.click();
	document.body.removeChild(a);
	URL.revokeObjectURL(url);
}

document.getElementById('assign-btn').addEventListener('click', () => {
	const woNumber = document.getElementById('wo-input').value.trim();
	if (!lastScannedBarcode || !woNumber) return;
	const part = findPartByBarcode(lastScannedBarcode);
	if (!part) return;
	const log = {
		timestamp: new Date().toLocaleString(),
		woNumber,
		barcode: lastScannedBarcode,
		partName: part.PartName,
		description: part.Description,
		stockLevel: part.StockLevel
	};
	logs.push(log);
	updateLogsTable();
});

document.getElementById('export-btn').addEventListener('click', exportLogsToCSV);

// html5-qrcode setup
function onScanSuccess(decodedText, decodedResult) {
	lastScannedBarcode = decodedText;
	document.getElementById('barcode-value').textContent = decodedText;
	const part = findPartByBarcode(decodedText);
	updatePartDetails(part);
}

function onScanFailure(error) {
	// Optionally handle scan errors
}

window.addEventListener('DOMContentLoaded', () => {
	const html5QrCode = new Html5Qrcode("qr-reader");
	Html5Qrcode.getCameras().then(cameras => {
		if (cameras && cameras.length) {
			html5QrCode.start(
				{ facingMode: "environment" },
				{ fps: 10, qrbox: 250 },
				onScanSuccess,
				onScanFailure
			);
		}
	});
});
